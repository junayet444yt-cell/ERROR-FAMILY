name: Build Family Shield apk

# This small file never needs to change. The real steps are in the normal folder ci/build.
# How to run: GitHub > Actions > "Build Family Shield apk" > Run workflow (give a release name, e.g. v1.0.0).
on:
  workflow_dispatch:
    inputs:
      tag:
        description: "Release name (for example v1.0.0)"
        default: "v1.0.0"

permissions:
  contents: write

env:
  GH_TOKEN: ${{ github.token }}
  TAG: ${{ github.event.inputs.tag }}

jobs:
  apk:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: ./ci/build
